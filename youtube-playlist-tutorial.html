跟著游標一步一步做：申請 API 金鑰、建立播放清單試算表、貼上程式碼、部署成網頁。

開場
按「播放」開始
    常見卡關提醒
    • 播放清單要設為「公開」或「不公開」；設成「私人」的清單，API 金鑰讀不到。
    • HTML 檔要命名為 Index，因為 Code.gs 用 createHtmlOutputFromFile('Index') 載入它。
    • 工作表名稱要叫「播放清單」；找不到時程式會改讀第一個工作表。
    • API 金鑰只貼在 Code.gs 第 9 行，不要放進網頁或傳給別人。
    • 修改程式後要「管理部署作業 → 編輯 → 新版本」，網址才會用到新程式。

/**
 * 從 Google Sheets 讀取所有播放清單
 */
function getPlaylistsFromSheet() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('播放清單');
    
    if (!sheet) {
      sheet = ss.getSheets()[0];
    }
    
    var data = sheet.getDataRange().getValues();
    
    if (data.length <= 1) {
      return { 
        ok: true, 
        playlists: [], 
        count: 0,
        message: '工作表中沒有播放清單資料'
      };
    }
    
    var playlists = [];
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      if (row[0] && row[1] && String(row[0]).trim() !== '' && String(row[1]).trim() !== '') {
        playlists.push({
          name: String(row[0]).trim(),
          playlistId: String(row[1]).trim(),
          description: row[2] ? String(row[2]).trim() : ''
        });
      }
    }
    
    return { 
      ok: true, 
      playlists: playlists, 
      count: playlists.length 
    };
    
  } catch (err) {
    return { 
      ok: false, 
      error: err.message, 
      playlists: [],
      count: 0
    };
  }
}

/**
 * 使用 YouTube Data API v3 取得播放清單所有影片
 * 支援分頁，可以取得完整播放清單
 */
function getPlaylistVideos(playlistId) {
  try {
    if (!playlistId) {
      return { 
        ok: false, 
        playlistId: '', 
        count: 0, 
        videos: [], 
        error: '播放清單 ID 不能為空' 
      };
    }

    // 檢查 API Key
    if (!YOUTUBE_API_KEY || YOUTUBE_API_KEY === '請在這裡填入你的 API Key') {
      return { 
        ok: false, 
        playlistId: '', 
        count: 0, 
        videos: [], 
        error: '請先在 Code.gs 中設定 YouTube API Key' 
      };
    }

    var pid = extractPlaylistId(playlistId);
    if (!pid) {
      return { 
        ok: false, 
        playlistId: '', 
        count: 0, 
        videos: [], 
        error: '無法解析播放清單 ID' 
      };
    }

    // 使用快取來避免重複請求
    var cache = CacheService.getScriptCache();
    var cacheKey = 'ytapi:' + pid;
    var cached = cache.get(cacheKey);
    
    if (cached) {
      return JSON.parse(cached);
    }

    // 取得所有影片（處理分頁）
    var videos = [];
    var nextPageToken = '';
    var maxResults = 50; // 每次請求最多 50 筆
    var totalFetched = 0;
    var maxPages = 20; // 最多取 20 頁（1000 支影片），避免超時
    var pageCount = 0;

    do {
      var url = 'https://www.googleapis.com/youtube/v3/playlistItems?' +
                'part=snippet&playlistId=' + pid +
                '&maxResults=' + maxResults +
                '&key=' + YOUTUBE_API_KEY;
      
      if (nextPageToken) {
        url += '&pageToken=' + nextPageToken;
      }

      var response = UrlFetchApp.fetch(url, {
        muteHttpExceptions: true,
        headers: {
          'Accept': 'application/json'
        }
      });

      if (response.getResponseCode() !== 200) {
        var errorText = response.getContentText();
        var errorMsg = 'API 請求失敗：HTTP ' + response.getResponseCode();
        
        try {
          var errorJson = JSON.parse(errorText);
          if (errorJson.error && errorJson.error.message) {
            errorMsg = errorJson.error.message;
          }
        } catch (e) {
          // 無法解析錯誤訊息，使用預設訊息
        }
        
        return { 
          ok: false, 
          playlistId: pid, 
          count: 0, 
          videos: [], 
          error: errorMsg
        };
      }

      var data = JSON.parse(response.getContentText());
      
      if (data.items && data.items.length > 0) {
        data.items.forEach(function(item) {
          if (item.snippet && item.snippet.resourceId && item.snippet.resourceId.videoId) {
            var videoId = item.snippet.resourceId.videoId;
            var title = item.snippet.title || '(無標題)';
            var author = item.snippet.videoOwnerChannelTitle || item.snippet.channelTitle || '';
            
            // 取得最佳品質的縮圖
            var thumbnail = '';
            if (item.snippet.thumbnails) {
              if (item.snippet.thumbnails.medium) {
                thumbnail = item.snippet.thumbnails.medium.url;
              } else if (item.snippet.thumbnails.default) {
                thumbnail = item.snippet.thumbnails.default.url;
              } else if (item.snippet.thumbnails.high) {
                thumbnail = item.snippet.thumbnails.high.url;
              }
            }
            
            if (!thumbnail) {
              thumbnail = 'https://i.ytimg.com/vi/' + videoId + '/mqdefault.jpg';
            }

            videos.push({
              videoId: videoId,
              title: title,
              author: author,
              thumbnail: thumbnail
            });
          }
        });
        
        totalFetched += data.items.length;
      }

      nextPageToken = data.nextPageToken || '';
      pageCount++;

      // 避免無限迴圈和超時
      if (pageCount >= maxPages) {
        Logger.log('已達到最大頁數限制：' + maxPages);
        break;
      }

    } while (nextPageToken);

    var result = { 
      ok: true, 
      playlistId: pid, 
      count: videos.length, 
      videos: videos,
      apiUsed: true,
      totalFetched: totalFetched
    };

    // 快取 10 分鐘
    cache.put(cacheKey, JSON.stringify(result), 600);
    
    return result;

  } catch (err) {
    Logger.log('錯誤: ' + err.message);
    return { 
      ok: false, 
      playlistId: '', 
      count: 0, 
      videos: [], 
      error: String(err.message || err) 
    };
  }
}

/**
 * 從輸入字串中提取播放清單 ID
 */
function extractPlaylistId(input) {
  if (!input) return '';
  var s = String(input).trim();

  // 直接是 PL 開頭的 ID
  if (s.match(/^PL[a-zA-Z0-9_-]+$/)) {
    return s;
  }

  // 從 URL 中提取 list 參數
  var match = s.match(/[?&]list=([a-zA-Z0-9_-]+)/);
  if (match && match[1]) {
    return match[1];
  }
  
  // 如果都不符合，嘗試找出 PL 開頭的部分
  var plMatch = s.match(/(PL[a-zA-Z0-9_-]+)/);
  if (plMatch && plMatch[1]) {
    return plMatch[1];
  }
  
  return '';
}

    Logger.log('❌ 請先設定 API Key');
    return false;

  var url = 'https://www.googleapis.com/youtube/v3/playlistItems?' +
            'part=snippet&playlistId=PLrAXtmErZgOeiKm4sgNOknGvNjby9efdf&maxResults=1&key=' + YOUTUBE_API_KEY;
  
  try {
    var response = UrlFetchApp.fetch(url, { muteHttpExceptions: true });

      return true;

      return false;
    }
  } catch (err) {
    Logger.log('❌ 測試 API Key 時發生錯誤: ' + err.message);
    return false;
  }

        titleDiv.className = 'video-title';
        titleDiv.textContent = (idx + 1) + '. ' + video.title;
        
        const authorDiv = document.createElement('div');
        authorDiv.className = 'video-author';
        authorDiv.textContent = video.author;
        
        infoDiv.appendChild(titleDiv);
        infoDiv.appendChild(authorDiv);
        
        li.appendChild(img);
        li.appendChild(infoDiv);
        li.onclick = () => playVideo(idx);
        
        if (idx === currentIndex) li.classList.add('active');
        
        videoListEl.appendChild(li);
      });
    }

    function playVideo(index) {
      currentIndex = index;
      const video = videos[index];
      
      videoTitleEl.textContent = (index + 1) + '. ' + video.title;
      embedErrorEl.classList.remove('show');
      
      const embedUrl = 'https://www.youtube.com/embed/' + video.videoId + '?autoplay=1&rel=0&playsinline=1';
      ytPlayerEl.src = embedUrl;
      
      // 更新列表中的 active 狀態
      const listItems = videoListEl.children;
      for (let i = 0; i < listItems.length; i++) {
        if (i === index) {
          listItems[i].classList.add('active');
        } else {
          listItems[i].classList.remove('active');
        }
      }
      
      actionsEl.style.display = 'flex';
      
      const videoUrl = 'https://www.youtube.com/watch?v=' + video.videoId;
      const playlistUrl = 'https://www.youtube.com/playlist?list=' + currentPlaylistId;
      
      openYtEl.href = videoUrl;
      openPlaylistEl.href = playlistUrl;
      errorOpenYtEl.href = videoUrl;
    }

    function escapeHtml(text) {
      const div = document.createElement('div');
      div.textContent = text;
      return div.innerHTML;
    }
